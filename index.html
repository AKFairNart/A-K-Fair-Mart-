-- A&K Fair Mart: strict listing edit/delete permissions -- Admin: can edit and delete every listing. -- Client: can edit and delete only listings where listing.user_id = auth.uid(). -- Run in Supabase Dashboard > SQL Editor. ALTER TABLE public.listing ENABLE ROW LEVEL SECURITY; -- Remove existing UPDATE/DELETE/ALL policies so an older permissive policy -- cannot accidentally let clients change or delete another client's listing. DO $$ DECLARE p record; BEGIN FOR p IN SELECT policyname FROM pg_policies WHERE schemaname = 'public' AND tablename = 'listing' AND cmd IN ('UPDATE', 'DELETE', 'ALL') LOOP EXECUTE format('DROP POLICY IF EXISTS %I ON public.listing', p.policyname); END LOOP; END $$; -- Only the signed-in owner or the designated admin may update a listing. CREATE POLICY "akfm_owner_or_admin_update_listing" ON public.listing FOR UPDATE TO authenticated USING ( user_id = auth.uid() OR lower(coalesce(auth.jwt() ->> 'email', '')) = lower('arbab.khalidrasheed@gmail.com') ) WITH CHECK ( user_id = auth.uid() OR lower(coalesce(auth.jwt() ->> 'email', '')) = lower('arbab.khalidrasheed@gmail.com') ); -- Only the signed-in owner or the designated admin may delete a listing. CREATE POLICY "akfm_owner_or_admin_delete_listing" ON public.listing FOR DELETE TO authenticated USING ( user_id = auth.uid() OR lower(coalesce(auth.jwt() ->> 'email', '')) = lower('arbab.khalidrasheed@gmail.com') ); -- INSERT and SELECT policies are intentionally left unchanged.
